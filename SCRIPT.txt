      <!-- FORMULARIO SOLO ADMIN -->
      <div id="formAdmin" class="hidden bg-slate-50 p-4 rounded-lg border mb-6 space-y-3">
        <h3 class="font-bold text-slate-700">Nuevo Registro de Suministro</h3>
        
        <div class="flex gap-2">
          <input type="text" id="regCodSuministro" placeholder="Cód. Suministro" class="w-full border p-2 rounded">
          <button onclick="buscarCod()" class="bg-gray-800 text-white px-3 py-2 rounded text-sm font-bold">Buscar</button>
        </div>

        <!-- Campos autocompletados -->
        <input type="text" id="regNombre" placeholder="Nombre Usuario" class="w-full border p-2 rounded bg-gray-100" readonly>
        <input type="text" id="regDomicilio" placeholder="Domicilio" class="w-full border p-2 rounded bg-gray-100" readonly>
        
        <div class="grid grid-cols-3 gap-2">
          <input type="text" id="regSector" placeholder="Sector" class="border p-2 rounded bg-gray-100" readonly>
          <input type="text" id="regCategoria" placeholder="Categoría" class="border p-2 rounded bg-gray-100" readonly>
          <input type="text" id="regMedidor" placeholder="Medidor" class="border p-2 rounded bg-gray-100" readonly>
        </div>

        <select id="regResponsable" class="w-full border p-2 rounded font-semibold text-slate-700">
          <option value="">-- Seleccionar Responsable --</option>
        </select>

        <input type="text" id="regMotivo" placeholder="Motivo de cambio" class="w-full border p-2 rounded">
        
        <button onclick="guardarRegistro()" class="w-full bg-green-600 text-white py-2 rounded font-bold">Guardar y Asignar</button>
      </div>

      <!-- LISTA DE REGISTROS -->
      <h3 class="font-bold text-slate-800 mb-2">Instalaciones Asignadas</h3>

  <script>
    // REEMPLAZA ESTA URL CON LA DE TU APPS SCRIPT IMPLEMENTADO EN EL PASO 3
    const API_URL = "URL_DE_TU_APPS_SCRIPT"const API_URL = "https://script.google.com/macros/s/AKfycbxNA-wPUmNt9v9_IyujkWrM1GD3OvslbM4QlGSuDH4e-WaSBc9af_liZIUAGCP0nJUG/exec";
    
    let usuarioActual = { dni: '', nombre: '', rol: '' };
    let listaResponsables = [];

    async function login() {
      const dni = document.getElementById('inputDni').value.trim();
      if (!dni) return alert('Ingrese su DNI');

      const res = await fetch(`${API_URL}?dni=${dni}`).then(r => r.json());
      if (!res.ok) return alert(res.mensaje);

      usuarioActual = { dni, nombre: res.nombre, rol: res.rol };
      listaResponsables = res.responsables || [];

      document.getElementById('sec-login').classList.add('hidden');
      document.getElementById('sec-panel').classList.remove('hidden');
      document.getElementById('lblUsuario').innerText = res.nombre;
      document.getElementById('lblRol').innerText = res.rol;

      if (res.rol === 'administrador') {
        document.getElementById('formAdmin').classList.remove('hidden');
        const select = document.getElementById('regResponsable');
        select.innerHTML = '<option value="">-- Seleccionar Responsable --</option>';
        listaResponsables.forEach(r => {
          select.innerHTML += `<option value="${r.dni}">${r.nombre}</option>`;
        });
      }

      renderRegistros(res.registros);
    }

    async function buscarCod() {
      const cod = document.getElementById('regCodSuministro').value.trim();
      if (!cod) return alert('Ingrese un código');

      const res = await fetch(API_URL, {
        method: 'POST',
        body: JSON.stringify({ dni: usuarioActual.dni, accion: 'buscarSuministro', codSuministro: cod })
      }).then(r => r.json());

      if (!res.ok) return alert(res.mensaje);

      document.getElementById('regNombre').value = res.nombreUsuario;
      document.getElementById('regDomicilio').value = res.domicilio;
      document.getElementById('regSector').value = res.sector;
      document.getElementById('regCategoria').value = res.categoria;
      document.getElementById('regMedidor').value = res.nroMedidor;
    }

    async function guardarRegistro() {
      const cod = document.getElementById('regCodSuministro').value.trim();
      const resp = document.getElementById('regResponsable').value;
      if (!cod || !resp) return alert('Busque un suministro y seleccione un responsable');

      const payload = {
        dni: usuarioActual.dni,
        accion: 'crearRegistro',
        codSuministro: cod,
        dniResponsable: resp,
        motivoCambio: document.getElementById('regMotivo').value
      };

      const res = await fetch(API_URL, { method: 'POST', body: JSON.stringify(payload) }).then(r => r.json());
      alert(res.mensaje);
      if (res.ok) login();
    }

    function renderRegistros(lista) {
      const cont = document.getElementById('listaRegistros');
      cont.innerHTML = '';

      if (lista.length === 0) {
        cont.innerHTML = '<p class="text-sm text-gray-500 text-center">No hay registros pendientes.</p>';
        return;
      }

      lista.forEach(reg => {
        const esPendiente = reg.estado === 'Pendiente';
        const card = document.createElement('div');
        card.className = "border rounded-lg p-4 bg-gray-50 space-y-2";
        
        card.innerHTML = `
          <div class="flex justify-between items-start">
            <div>
              <p class="font-bold text-slate-800">${reg.codSuministro} - ${reg.nombreUsuario}</p>
              <p class="text-xs text-gray-600">${reg.domicilio} (Medidor: ${reg.nroMedidor})</p>
            </div>
            <span class="text-xs px-2 py-1 rounded font-bold ${esPendiente ? 'bg-amber-100 text-amber-800' : 'bg-green-100 text-green-800'}">${reg.estado}</span>
          </div>

          ${esPendiente ? `
            <div class="border-t pt-2 mt-2 space-y-2">
              <label class="block text-xs font-bold text-gray-700">Tomar foto o adjuntar PDF:</label>
              <!-- capture="environment" abre la cámara trasera en smartphones -->
              <input type="file" id="file-${reg.id}" accept="image/*,application/pdf" capture="environment" class="w-full text-xs">
              <input type="text" id="obs-${reg.id}" placeholder="Observación opcional" class="w-full border p-1 rounded text-xs">
              <button onclick="ejecutarTapa('${reg.id}')" class="w-full bg-blue-600 text-white py-1.5 rounded font-bold text-sm">Validar Instalación</button>
            </div>
          ` : `
            <p class="text-xs text-gray-500">Ejecutado el: ${reg.fechaEjecucion}</p>
          `}
        `;
        cont.appendChild(card);
      });
    }

    async function ejecutarTapa(id) {
      const fileInput = document.getElementById(`file-${id}`);
      const obs = document.getElementById(`obs-${id}`).value;
      const archivos = [];

      if (fileInput.files.length > 0) {
        const file = fileInput.files[0];
        const base64 = await toBase64(file);
        archivos.push({ nombre: file.name, mime: file.type, base64 });
      }

      const payload = {
        dni: usuarioActual.dni,
        accion: 'marcarEjecutado',
        registroId: id,
        comentario: obs,
        archivos
      };

      const res = await fetch(API_URL, { method: 'POST', body: JSON.stringify(payload) }).then(r => r.json());
      alert(res.mensaje);
      if (res.ok) login();
    }

    const toBase64 = file => new Promise((res, rej) => {
      const reader = new FileReader();
      reader.readAsDataURL(file);
      reader.onload = () => res(reader.result);
      reader.onerror = err => rej(err);
    });
  </script>
